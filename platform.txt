# NOTE: -fno-extern-tls-init prevents the C++ frontend from emitting guarded
# references to the thread-local init function (e.g. _ZTH5errno) for extern
# thread_local variables such as picolibc's `errno`. These variables are
# statically initialized and need no dynamic init, but with the default
# -fextern-tls-init the compiler emits a weak `_ZTH<var>` reference that the
# llext loader cannot process, failing with "Undefined symbol ... _ZTH5errno"
# (ret -61). The toolchain 1.0.1 update made this surface for std::stoi/std::stol
# and any other libstdc++ code that touches errno (e.g. Arduino_RouterBridge).
compiler.zephyr.common_cxxflags=-fno-exceptions -fno-rtti -fdata-sections -ffunction-sections -fno-unwind-tables -fno-extern-tls-init
# NOTE: -lstdc++/-lsupc++ are intentionally NOT listed here. They are linked
# inside the --start-group of build.link_command (see compiler.zephyr.extra_ldflags)
# so that the core archive is scanned before libstdc++. Together with
# -Wl,-u,__cxa_allocate_exception this guarantees the core's exception-ABI stubs
# (cores/arduino/abi.cpp) win over libstdc++'s eh_*.o, keeping the C++ exception
# machinery and the libgcc unwinder (whose __gnu_Unwind_Find_exidx / __exidx_*
# references cannot be resolved by the llext loader) out of the sketch llext.
compiler.zephyr.common_ldflags=-fno-exceptions -fno-rtti -fno-threadsafe-statics -fno-unwind-tables -fno-use-cxa-atexit -lnosys -nostdlib -Wl,-u,__cxa_allocate_exception

compiler.zephyr.extra_ldflags=-lstdc++ -lsupc++ -lm -lgcc -lc

recipe.hooks.prebuild.1.pattern=sh -c 'if [ ! -f "{build.variant.path}/includes.txt" ]; then echo "Error: No binaries found for {_id}. See README.md to compile or download them first." && exit 1; fi'

## Compile c files
recipe.c.o.pattern="{compiler.path}{compiler.c.cmd}" {compiler.c.flags} -DARDUINO={runtime.ide.version} -DARDUINO_{build.board} -DARDUINO_ARCH_{build.arch} -DARDUINO_ARCH_ZEPHYR {build.library_discovery_phase_flag} {build.extra_flags} {build.extra_ldflags} {compiler.c.extra_flags} "-I{build.core.path}/api/deprecated" "-I{build.core.path}/api/deprecated-avr-comp" {includes} "-iprefix{build.variant.path}" "@{compiler.zephyr.includes_file}" -o "{object_file}" "{source_file}"

recipe.cpp.o.pattern="{compiler.path}{compiler.cpp.cmd}" {compiler.cpp.flags} -DARDUINO={runtime.ide.version} -DARDUINO_{build.board} -DARDUINO_ARCH_{build.arch} -DARDUINO_ARCH_ZEPHYR {build.library_discovery_phase_flag} {includes} {build.extra_flags} {build.extra_ldflags} {compiler.cpp.extra_flags} "-I{build.core.path}/api/deprecated" "-I{build.core.path}/api/deprecated-avr-comp" "-iprefix{build.variant.path}" "@{compiler.zephyr.includes_file}" "{source_file}" -o "{object_file}"

# Expects a {loader.cmd} and a {sketch.cmd} to be defined in the tool's
# section. The loader is flashed if it does not match the current (maybe
# board-specific) core version. The sketch is then flashed in any case.
# Extra arguments are passed to either command depending on whether the other
# part will be flashed in the same session or not.
empty=
loader.args.with_sketch={empty}
loader.args.without_sketch={empty}
sketch.args.with_loader={empty}
sketch.args.without_loader={empty}
flash-loader-condition=not eq "v{upload.port.properties.configuration}" "v{version}"
flash-loader-and-sketch.1=if {flash-loader-condition} {loader.cmd}{loader.args.with_sketch}
flash-loader-and-sketch.2=if {flash-loader-condition} {sketch.cmd}{sketch.args.with_loader}
flash-loader-and-sketch.3=if not {flash-loader-condition} {sketch.cmd}{sketch.args.without_loader}
tools.patterns.flash-loader-and-sketch="{runtime.tools.scripting-tools.path}/scripting-tools" {flash-loader-and-sketch.1} :: {flash-loader-and-sketch.2} :: {flash-loader-and-sketch.3}
tools.patterns.flash-loader-only={loader.cmd}{loader.args.without_sketch}
